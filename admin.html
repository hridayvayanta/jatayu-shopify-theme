1-Click Supabase Table Setup SQL

Paste this into the **SQL Editor** in your Supabase dashboard to create the tables in 5 seconds: 

create table if not exists products (
  id bigint primary key,
  title text not null,
  category text,
  price numeric,
  image text,
  secondary_image text,
  badge text,
  sizes text[],
  description text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);

create table if not exists orders (
  id text primary key,
  customer_name text,
  customer_email text,
  customer_phone text,
  total_price numeric,
  items text,
  payment_method text,
  payment_id text,
  status text,
  created_at timestamp with time zone default timezone('utc'::text, now())
);
